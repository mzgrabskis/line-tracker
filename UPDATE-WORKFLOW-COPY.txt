name: Update data
on:
  schedule:
    - cron: "0 */4 * * *"      # every 4 hours
  workflow_dispatch:            # lets you press "Run workflow" by hand
permissions:
  contents: write
concurrency: update-data
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - run: pip install -r requirements.txt
      - uses: actions/cache@v4
        with:
          path: cache
          key: el-cache-${{ github.run_id }}
          restore-keys: el-cache-
      - run: python build_data.py
      - name: Publish data
        if: always()
        run: |
          git config user.name "data-bot"
          git config user.email "data-bot@users.noreply.github.com"
          git add docs/data
          git diff --cached --quiet && exit 0
          if git log -1 --pretty=%s | grep -q "^data update"; then
            git commit --amend -m "data update" && git push --force
          else
            git commit -m "data update" && git push
          fi
